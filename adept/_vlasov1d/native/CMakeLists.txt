cmake_minimum_required(VERSION 3.24)
project(adept_vlasov_native LANGUAGES CXX)
option(VLASOV_CUDA "Use CUDA/cuFFT (OFF builds the FFTW validation backend)" ON)
find_package(MPI REQUIRED COMPONENTS CXX)
add_library(adept_vlasov SHARED api.cpp core/backend.cpp)
target_compile_features(adept_vlasov PRIVATE cxx_std_17)
target_link_libraries(adept_vlasov PRIVATE MPI::MPI_CXX)
set_target_properties(adept_vlasov PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(VLASOV_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set_source_files_properties(core/backend.cpp PROPERTIES LANGUAGE CUDA)
  set_target_properties(adept_vlasov PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES 80)
  target_compile_definitions(adept_vlasov PRIVATE VLASOV_CUDA)
  target_compile_options(adept_vlasov PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda>)
  target_link_libraries(adept_vlasov PRIVATE CUDA::cudart CUDA::cufft)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FFTW REQUIRED IMPORTED_TARGET fftw3)
  target_link_libraries(adept_vlasov PRIVATE PkgConfig::FFTW)
endif()
