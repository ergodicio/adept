cmake_minimum_required(VERSION 3.24)
project(vlasov_split_benchmark LANGUAGES CXX)

option(VLASOV_CUDA "Build the CUDA/cuFFT backend (OFF uses FFTW for validation)" ON)
find_package(MPI REQUIRED COMPONENTS CXX)
add_executable(vlasov_bench main.cpp backend.cpp)
target_compile_features(vlasov_bench PRIVATE cxx_std_17)
target_link_libraries(vlasov_bench PRIVATE MPI::MPI_CXX)
if(VLASOV_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set_source_files_properties(backend.cpp PROPERTIES LANGUAGE CUDA)
  set_target_properties(vlasov_bench PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES 80)
  target_compile_definitions(vlasov_bench PRIVATE VLASOV_CUDA)
  target_compile_options(vlasov_bench PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda>)
  target_link_libraries(vlasov_bench PRIVATE CUDA::cudart CUDA::cufft)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FFTW REQUIRED IMPORTED_TARGET fftw3)
  target_link_libraries(vlasov_bench PRIVATE PkgConfig::FFTW)
endif()
